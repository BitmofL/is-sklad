-- Создание таблицы "КатегорииТоваров"
CREATE TABLE КатегорииТоваров (
    ID_Категории INT PRIMARY KEY IDENTITY(1,1),
    Наименование NVARCHAR(255) NOT NULL,
    Описание NVARCHAR(MAX) NULL
);

-- Создание таблицы "Товары"
CREATE TABLE Товары (
    ID_Товара INT PRIMARY KEY IDENTITY(1,1),
    Наименование NVARCHAR(255) NOT NULL,
    Описание NVARCHAR(MAX) NULL,
    Цена DECIMAL(18, 2) NOT NULL,
    ЕдИзм NVARCHAR(50) NOT NULL,
    ID_Категории INT NOT NULL,
    FOREIGN KEY (ID_Категории) REFERENCES КатегорииТоваров(ID_Категории)
);

-- Создание таблицы "Поставщики"
CREATE TABLE Поставщики (
    ID_Поставщика INT PRIMARY KEY IDENTITY(1,1),
    Наименование NVARCHAR(255) NOT NULL,
    Адрес NVARCHAR(255) NULL,
    Телефон NVARCHAR(20) NULL,
    Email NVARCHAR(255) NULL
);

-- Создание таблицы "СкладскиеОстатки"
CREATE TABLE СкладскиеОстатки (
    ID_Склад INT NOT NULL,
    ID_Товара INT NOT NULL,
    Количество INT NOT NULL DEFAULT 0,
    PRIMARY KEY (ID_Склад, ID_Товара),
    FOREIGN KEY (ID_Товара) REFERENCES Товары(ID_Товара)
);

-- Создание таблицы "Склады"
CREATE TABLE Склады (
    ID_Склада INT PRIMARY KEY IDENTITY (1,1),
    Наименование NVARCHAR(255) NOT NULL,
    Адрес NVARCHAR(255) NULL
);


-- Создание таблицы "Клиенты"
CREATE TABLE Клиенты (
    ID_Клиента INT PRIMARY KEY IDENTITY(1,1), -- Автоинкрементный первичный ключ
    Имя NVARCHAR(100) NOT NULL,       -- Имя клиента (не может быть пустым)
    Адрес NVARCHAR(255),                -- Адрес клиента
    Телефон NVARCHAR(20)                 -- Телефон клиента
);

CREATE TABLE Отгрузки (
    ID_Отгрузки INT PRIMARY KEY IDENTITY(1,1),
    ID_Склада INT NOT NULL,
    ID_Клиента INT NOT NULL,
    ID_Товара INT NOT NULL,
    Количество INT NOT NULL,
    ДатаОтгрузки DATETIME NOT NULL,
    FOREIGN KEY (ID_Склада) REFERENCES Склады(ID_Склада),
    FOREIGN KEY (ID_Клиента) REFERENCES Клиенты(ID_Клиента),
    FOREIGN KEY (ID_Товара) REFERENCES Товары(ID_Товара)
);
